export default function App() {
  const [user, setUser] = useState(null);
  const [screen, setScreen] = useState("login");

  const handleLogin = (u) => {
    setUser(u);
    setScreen(u.role === "admin" ? "admin" : "checklist");
  };

  const handleLogout = () => { setUser(null); setScreen("login"); };

  if (screen === "login") return <Login onLogin={handleLogin} />;
  if (screen === "admin") return <AdminPanel user={user} onLogout={handleLogout} />;
  return <ChecklistFlow user={user} onLogout={handleLogout} />;
}

// ─────────────────────────────────────────────
// LOGIN
// ─────────────────────────────────────────────
function Login({ onLogin }) {
  const [usuario, setUsuario] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(false);

  const login = async () => {
    if (!usuario.trim() || !password.trim()) return;
    setLoading(true);
    setError("");
    try {
      if (DEMO_MODE) {
        const found = DEMO_USERS.find(u => u.nombre.toLowerCase() === usuario.toLowerCase() && u.password === password);
        if (!found) throw new Error("Usuario o contraseña incorrectos");
        onLogin(found);
      } else {
        const data = await sb(`usuarios?nombre=eq.${encodeURIComponent(usuario)}&password=eq.${encodeURIComponent(password)}&select=*`);
        if (!data || data.length === 0) throw new Error("Usuario o contraseña incorrectos");
        onLogin(data[0]);
      }
    } catch (e) {
      setError(e.message);
    }
    setLoading(false);
  };

  return (
    <div style={S.bg}>
      {DEMO_MODE && (
        <div style={S.demoBanner}>
          🔧 <strong>Modo demo</strong> — Conectá Supabase para usar en producción
        </div>
      )}
      <div style={S.card}>
        <div style={{ textAlign: "center", marginBottom: 28 }}>
          <div style={{ fontSize: 56 }}>🏭</div>
          <h1 style={S.titulo}>FábricaCheck</h1>
          <p style={S.sub}>Sistema de cierre de turno</p>
        </div>
        {DEMO_MODE && (
          <div style={S.demoHint}>
            <strong>Demo:</strong> Admin → usuario: <code>Admin</code> / pass: <code>admin123</code><br />
            Operario → usuario: <code>Juan Pérez</code> / pass: <code>1234</code>
          </div>
        )}
        <label style={S.label}>Usuario</label>
        <input style={S.input} placeholder="Tu nombre de usuario" value={usuario} onChange={e => setUsuario(e.target.value)} />
        <label style={{ ...S.label, marginTop: 12 }}>Contraseña</label>
        <input style={S.input} type="password" placeholder="••••••" value={password} onChange={e => setPassword(e.target.value)} onKeyDown={e => e.key === "Enter" && login()} />
        {error && <div style={S.errorMsg}>{error}</div>}
        <button style={{ ...S.btnPrimary, marginTop: 20, opacity: loading ? 0.6 : 1 }} onClick={login} disabled={loading}>
          {loading ? "Ingresando..." : "Ingresar →"}
        </button>
      </div>
    </div>
  );
}

// CHECKLIST FLOW (OPERARIO)
// ─────────────────────────────────────────────
function ChecklistFlow({ user, onLogout }) {
  const [step, setStep] = useState("inicio"); // inicio | checklist | informe
  const [current, setCurrent] = useState(0);
  const [items, setItems] = useState([]);
  const [horaInicio, setHoraInicio] = useState(null);
  const [guardando, setGuardando] = useState(false);
  const [informeGuardado, setInformeGuardado] = useState(null);

  const iniciar = () => {
    setItems(CHECKLIST_ITEMS.map(i => ({ ...i, status: "pendiente", timestamp: null })));
    setHoraInicio(horaAhora());
    setCurrent(0);
    setStep("checklist");
  };

  const responder = async (status) => {
    const updated = items.map((item, idx) =>
      idx === current ? { ...item, status, timestamp: horaAhora() } : item
    );
    setItems(updated);
    if (current < items.length - 1) {
      setCurrent(current + 1);
    } else {
      // Guardar informe
      setGuardando(true);
      const informe = {
        operario_id: user.id,
        operario_nombre: user.nombre,
        fecha: new Date().toISOString(),
        hora_inicio: horaInicio,
        hora_fin: horaAhora(),
        items: updated,
        total_realizados: updated.filter(i => i.status === "hecho").length,
        total_no_realizados: updated.filter(i => i.status === "nohecho").length,
      };
      if (DEMO_MODE) {
        demoInformes = [{ id: Date.now(), ...informe }, ...demoInformes];
        setInformeGuardado(informe);
      } else {
        try {
          await sb("informes", { method: "POST", body: JSON.stringify(informe) });
          setInformeGuardado(informe);
        } catch (e) {
          setInformeGuardado(informe);
        }
      }
      setGuardando(false);
      setStep("informe");
    }
  };

  if (step === "inicio") {
    return (
      <div style={S.bg}>
        <div style={S.card}>
          <div style={S.topBar}>
            <span style={S.chip}>👷 {user.nombre}</span>
            <button style={S.btnLogout} onClick={onLogout}>Salir</button>
          </div>
          <div style={{ textAlign: "center", margin: "24px 0 20px" }}>
            <div style={{ fontSize: 52 }}>📋</div>
            <h2 style={S.titulo}>Checklist de Cierre</h2>
            <p style={S.sub}>{fechaHoy()}</p>
          </div>
          <div style={S.infoBox}>
            <div style={S.infoBoxTitle}>Este turno revisarás:</div>
            {CHECKLIST_ITEMS.map(item => (
              <div key={item.id} style={S.previewItem}>
                <span>{item.icono}</span>
                <span>{item.tarea}</span>
              </div>
            ))}
          </div>
          <button style={{ ...S.btnPrimary, marginTop: 20 }} onClick={iniciar}>
            Iniciar checklist →
          </button>
        </div>
      </div>
    );
  }

  if (step === "checklist") {
    const item = items[current];
    const completados = items.filter(i => i.status !== "pendiente").length;
    return (
      <div style={S.bg}>
        <div style={S.card}>
          <div style={S.topBar}>
            <span style={S.chip}>👷 {user.nombre}</span>
            <span style={{ ...S.chip, background: "#1a1a2e", color: "#fff" }}>{current + 1}/{items.length}</span>
          </div>
          <div style={S.progressBg}>
            <div style={{ ...S.progressFill, width: `${(completados / items.length) * 100}%` }} />
          </div>
          <div style={S.miniRow}>
            {items.map((it, idx) => (
              <div key={it.id} style={{
                ...S.miniDot,
                background: idx === current ? "#FFC107" : it.status === "hecho" ? "#28a745" : it.status === "nohecho" ? "#dc3545" : "#e0e0e0",
                transform: idx === current ? "scale(1.3)" : "scale(1)",
              }} />
            ))}
          </div>
          <div style={S.tareaBox}>
            <div style={{ fontSize: 52, marginBottom: 8 }}>{item.icono}</div>
            <div style={S.tareaZona}>{item.zona}</div>
            <div style={S.tareaTitulo}>{item.tarea}</div>
            <p style={S.tareaDesc}>{item.descripcion}</p>
          </div>
          <div style={S.botonesRow}>
            <button style={S.btnNo} onClick={() => responder("nohecho")}>✗ No realizado</button>
            <button style={S.btnSi} onClick={() => responder("hecho")}>✓ Realizado</button>
          </div>
        </div>
      </div>
    );
  }

  if (step === "informe") {
    const hechos = items.filter(i => i.status === "hecho").length;
    const noHechos = items.filter(i => i.status === "nohecho").length;
    const todoOk = noHechos === 0;
    return (
      <div style={S.bg}>
        <div style={{ ...S.card, maxWidth: 440 }}>
          {guardando ? (
            <div style={{ textAlign: "center", padding: 40 }}>
              <div style={{ fontSize: 40 }}>⏳</div>
              <p>Guardando informe...</p>
            </div>
          ) : (
            <>
              <div style={{ textAlign: "center", marginBottom: 20 }}>
                <div style={{ fontSize: 52 }}>{todoOk ? "🎉" : "⚠️"}</div>
                <h2 style={S.titulo}>{todoOk ? "¡Turno cerrado!" : "Cierre con observaciones"}</h2>
                <p style={S.sub}>{fechaHoy()}</p>
                <div style={{ ...S.chip, display: "inline-block", marginTop: 8, background: "#e8f5e9", color: "#2e7d32" }}>
                  ✅ Informe guardado
                </div>
              </div>
              <div style={S.resumenRow}>
                <div style={{ ...S.resumenBox, background: "#e8f5e9" }}>
                  <div style={S.resumenNum}>{hechos}</div>
                  <div style={S.resumenLabel}>Realizadas</div>
                </div>
                <div style={{ ...S.resumenBox, background: noHechos > 0 ? "#ffebee" : "#f5f5f5" }}>
                  <div style={S.resumenNum}>{noHechos}</div>
                  <div style={S.resumenLabel}>No realizadas</div>
                </div>
              </div>
              <div style={S.detalleSec}>
                {items.map(item => (
                  <div key={item.id} style={S.detalleRow}>
                    <span style={{ fontSize: 20 }}>{item.icono}</span>
                    <span style={{ flex: 1, fontSize: 14 }}>{item.tarea}</span>
                    <span style={{
                      ...S.badge,
                      background: item.status === "hecho" ? "#e8f5e9" : "#ffebee",
                      color: item.status === "hecho" ? "#2e7d32" : "#c62828",
                    }}>
                      {item.status === "hecho" ? "✓ Hecho" : "✗ Pendiente"}
                    </span>
                  </div>
                ))}
              </div>
              <button style={S.btnPrimary} onClick={() => setStep("inicio")}>Nuevo turno</button>
              <button style={{ ...S.btnSecondary, marginTop: 8 }} onClick={onLogout}>Cerrar sesión</button>
            </>
          )}
        </div>
      </div>
    );
  }
}

// ADMIN PANEL
// ─────────────────────────────────────────────
function AdminPanel({ user, onLogout }) {
  const [informes, setInformes] = useState([]);
  const [loading, setLoading] = useState(true);
  const [selected, setSelected] = useState(null);
  const [filtroOp, setFiltroOp] = useState("todos");

  useEffect(() => {
    cargarInformes();
  }, []);

  const cargarInformes = async () => {
    setLoading(true);
    try {
      if (DEMO_MODE) {
        setInformes(demoInformes);
      } else {
        const data = await sb("informes?order=fecha.desc&select=*");
        setInformes(data || []);
      }
    } catch (e) {
      setInformes([]);
    }
    setLoading(false);
  };

  const operarios = ["todos", ...new Set(informes.map(i => i.operario_nombre))];
  const filtrados = filtroOp === "todos" ? informes : informes.filter(i => i.operario_nombre === filtroOp);

  if (selected) {
    return <InformeDetalle informe={selected} onBack={() => setSelected(null)} />;
  }

  return (
    <div style={S.bg}>
      <div style={{ ...S.card, maxWidth: 500 }}>
        <div style={S.topBar}>
          <span style={{ ...S.chip, background: "#fff3cd", color: "#856404" }}>👑 {user.nombre}</span>
          <div style={{ display: "flex", gap: 8 }}>
            <button style={S.btnRefresh} onClick={cargarInformes}>↻</button>
            <button style={S.btnLogout} onClick={onLogout}>Salir</button>
          </div>
        </div>

        <h2 style={{ ...S.titulo, marginTop: 20, marginBottom: 4 }}>Panel Admin</h2>
        <p style={S.sub}>Informes de cierre de turno</p>

        {/* Estadísticas rápidas */}
        <div style={S.statsRow}>
          <div style={S.statBox}>
            <div style={S.statNum}>{informes.length}</div>
            <div style={S.statLabel}>Turnos totales</div>
          </div>
          <div style={S.statBox}>
            <div style={S.statNum}>{operarios.length - 1}</div>
            <div style={S.statLabel}>Operarios</div>
          </div>
          <div style={S.statBox}>
            <div style={S.statNum}>
              {informes.length > 0
                ? Math.round(informes.reduce((acc, i) => acc + (i.total_realizados / (i.total_realizados + i.total_no_realizados)) * 100, 0) / informes.length)
                : 0}%
            </div>
            <div style={S.statLabel}>Cumplimiento</div>
          </div>
        </div>

        {/* Filtro por operario */}
        <div style={{ marginBottom: 16 }}>
          <label style={S.label}>Filtrar por operario</label>
          <select style={S.select} value={filtroOp} onChange={e => setFiltroOp(e.target.value)}>
            {operarios.map(op => <option key={op} value={op}>{op === "todos" ? "Todos los operarios" : op}</option>)}
          </select>
        </div>

        {/* Lista de informes */}
        {loading ? (
          <div style={{ textAlign: "center", padding: 32, color: "#888" }}>Cargando informes...</div>
        ) : filtrados.length === 0 ? (
          <div style={S.emptyState}>
            <div style={{ fontSize: 40 }}>📭</div>
            <p>No hay informes todavía.</p>
            {DEMO_MODE && <p style={{ fontSize: 12, color: "#aaa" }}>Completá un checklist como operario para ver el resultado aquí.</p>}
          </div>
        ) : (
          <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
            {filtrados.map(inf => (
              <div key={inf.id} style={S.informeCard} onClick={() => setSelected(inf)}>
                <div style={{ display: "flex", justifyContent: "space-between", alignItems: "flex-start" }}>
                  <div>
                    <div style={{ fontWeight: 700, color: "#1a1a2e", fontSize: 15 }}>👷 {inf.operario_nombre}</div>
                    <div style={{ fontSize: 12, color: "#888", marginTop: 2 }}>
                      {formatFecha(inf.fecha)} · {formatHora(inf.fecha)}
                    </div>
                  </div>
                  <div style={{
                    ...S.badge,
                    background: inf.total_no_realizados === 0 ? "#e8f5e9" : "#ffebee",
                    color: inf.total_no_realizados === 0 ? "#2e7d32" : "#c62828",
                    fontSize: 12,
                  }}>
                    {inf.total_no_realizados === 0 ? "✓ Completo" : `⚠️ ${inf.total_no_realizados} pendiente${inf.total_no_realizados > 1 ? "s" : ""}`}
                  </div>
                </div>
                <div style={{ display: "flex", gap: 12, marginTop: 10 }}>
                  <span style={{ fontSize: 12, color: "#555" }}>✅ {inf.total_realizados} realizadas</span>
                  <span style={{ fontSize: 12, color: "#555" }}>❌ {inf.total_no_realizados} no realizadas</span>
                </div>
              </div>
            ))}
          </div>
        )}
      </div>
    </div>
  );
}
